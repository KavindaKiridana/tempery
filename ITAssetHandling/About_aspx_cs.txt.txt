        public void calculateTotal(string txtQtyId, string txtPriceId, string lblTotal)
        {
            // Get the ContentPlaceHolder
            ContentPlaceHolder mainContent = (ContentPlaceHolder)Master.FindControl("MainContent");

            if (mainContent == null) return;

            // Find the controls inside the placeholder
            TextBox qtyTextBox = (TextBox)mainContent.FindControl(txtQtyId);
            TextBox priceTextBox = (TextBox)mainContent.FindControl(txtPriceId);
            Label totalLabel = (Label)mainContent.FindControl(lblTotal);

            if (qtyTextBox == null || priceTextBox == null || totalLabel == null)
            {
                return;
            }

            if (string.IsNullOrEmpty(qtyTextBox.Text) || string.IsNullOrEmpty(priceTextBox.Text))
            {
                totalLabel.Text = "";
                return;
            }

            decimal qty;
            decimal price;

            if (!decimal.TryParse(qtyTextBox.Text, out qty))
            {
                totalLabel.Text = "Quantity must be a number";
                return;
            }

            if (!decimal.TryParse(priceTextBox.Text, out price))
            {
                totalLabel.Text = "Price must be a number";
                return;
            }

            totalLabel.Text = (qty * price).ToString("N2"); // formats to 2 decimal places
            finalCost();
        }

        //function for templates list
        private void LoadTemplates(decimal totalCost)
        {
            string currencyForTemplate = lblCurrency.Text.ToString();

            // Check if company is selected
            if (ddlCompany.SelectedValue == null || string.IsNullOrEmpty(ddlCompany.SelectedValue) || ddlCompany.SelectedValue == "0")
            {
                ddlTemplate.Items.Clear();
                ddlTemplate.Items.Insert(0, new ListItem("-- Select Company First --", "0"));
                return;
            }

            int selectedCompany = Convert.ToInt32(ddlCompany.SelectedValue);
            checkconnection();

            try
            {
                sqlconn.Open();

                if (currencyForTemplate == "USD")
                {
                    // For USD, load all active templates for the company
                    // string query = "SELECT TemplateId, IsMDSign, ITManagerId, CEOId, MDId FROM Template WHERE IsActive=1 AND CompanyId=@CompanyId ORDER BY TemplateId";
                    string query = @"
SELECT
    t.TemplateId,
    t.IsMDSign,
    t.ITManagerId,
    IT.FullName AS ITManagerName,
    t.CEOId,
    CEO.FullName AS CEOName,
    t.MDId,
    MD.FullName AS MDName
FROM Template t
INNER JOIN Users IT ON t.ITManagerId = IT.UsersId
INNER JOIN Users CEO ON t.CEOId = CEO.UsersId
LEFT JOIN Users MD ON t.MDId = MD.UsersId
WHERE t.IsActive = 1 AND t.CompanyId = @CompanyId
ORDER BY t.TemplateId;";
                    SqlCommand cmd = new SqlCommand(query, sqlconn);
                    cmd.Parameters.AddWithValue("@CompanyId", selectedCompany);
                    SqlDataReader reader = cmd.ExecuteReader();

                    // Clear existing items and add default option
                    ddlTemplate.Items.Clear();
                    // ddlTemplate.Items.Insert(0, new ListItem("-- Select Template --", "0"));

                    while (reader.Read())
                    {
                        string templateId = reader["TemplateId"].ToString();
                        string displayText = reader["CEOName"].ToString() + "," + reader["ITManagerName"].ToString() + "," + reader["MDName"].ToString();
                        ddlTemplate.Items.Add(new ListItem(displayText, templateId));
                    }
                    reader.Close();
                }
                else if (currencyForTemplate == "LKR")
                {
                    // Use the totalCost parameter passed to the function instead of parsing from label
                    if (totalCost > 15000.0m)
                    {
                        // Cost is larger than 15000, so MD would sign here
                        //string query = "SELECT TemplateId, IsMDSign, ITManagerId, CEOId, MDId FROM Template WHERE IsActive=1 AND CompanyId=@CompanyId AND IsMDSign=1 ORDER BY TemplateId";
                        string query = @"
SELECT 
    t.TemplateId,
    t.IsMDSign,
    t.ITManagerId,
    IT.FullName AS ITManagerName,
    t.CEOId,
    CEO.FullName AS CEOName,
    t.MDId,
    MD.FullName AS MDName
FROM Template t
INNER JOIN Users IT ON t.ITManagerId = IT.UsersId
INNER JOIN Users CEO ON t.CEOId = CEO.UsersId
LEFT JOIN Users MD ON t.MDId = MD.UsersId
WHERE t.IsActive = 1 
  AND t.IsMDSign = 1 
  AND t.CompanyId = @CompanyId
ORDER BY t.TemplateId";
                        SqlCommand cmd = new SqlCommand(query, sqlconn);
                        cmd.Parameters.AddWithValue("@CompanyId", selectedCompany);
                        SqlDataReader reader = cmd.ExecuteReader();

                        // Clear existing items and add default option
                        ddlTemplate.Items.Clear();
                        //  ddlTemplate.Items.Insert(0, new ListItem("-- Select Template (MD Sign Required) --", "0"));

                        while (reader.Read())
                        {
                            string templateId = reader["TemplateId"].ToString();
                            string displayText = reader["CEOName"].ToString() + "," + reader["ITManagerName"].ToString() + "," + reader["MDName"].ToString();
                            ddlTemplate.Items.Add(new ListItem(displayText, templateId));
                        }
                        reader.Close();
                    }
                    else // totalCost <= 15000.0m ,MD wouldn't sign
                    {
                        // Cost is small or equal to 15000, so other authority
                        //string query = "SELECT TemplateId, IsMDSign, ITManagerId, CEOId, MDId FROM Template WHERE IsActive=1 AND CompanyId=@CompanyId AND IsMDSign=0 ORDER BY TemplateId";
                        string query = @"
SELECT 
    t.TemplateId,
    t.IsMDSign,
    t.ITManagerId,
    IT.FullName AS ITManagerName,
    t.CEOId,
    CEO.FullName AS CEOName,
    t.MDId,
    MD.FullName AS MDName
FROM Template t
INNER JOIN Users IT ON t.ITManagerId = IT.UsersId
INNER JOIN Users CEO ON t.CEOId = CEO.UsersId
LEFT JOIN Users MD ON t.MDId = MD.UsersId
WHERE t.IsActive = 1 
  AND t.IsMDSign = 0 
  AND t.CompanyId = @CompanyId
ORDER BY t.TemplateId";
                        SqlCommand cmd = new SqlCommand(query, sqlconn);
                        cmd.Parameters.AddWithValue("@CompanyId", selectedCompany);
                        SqlDataReader reader = cmd.ExecuteReader();

                        // Clear existing items and add default option
                        ddlTemplate.Items.Clear();
                        // ddlTemplate.Items.Insert(0, new ListItem("-- Select Template (MD wouldn't sign) --", "0"));

                        while (reader.Read())
                        {
                            string templateId = reader["TemplateId"].ToString();
                            //string displayText = $"Template {templateId} (MD wouldn't sign)";
                            string displayText = reader["CEOName"].ToString() + "," + reader["ITManagerName"].ToString() + "," + reader["MDName"].ToString();
                            ddlTemplate.Items.Add(new ListItem(displayText, templateId));
                        }
                        reader.Close();
                    }
                }
                else
                {
                    // Handle unknown currency
                    ddlTemplate.Items.Clear();
                    ddlTemplate.Items.Insert(0, new ListItem("-- Unknown Currency Type --", "0"));
                    Response.Write("Warning: Unknown currency type: " + currencyForTemplate);
                }
            }
            catch (Exception ex)
            {
                // Handle error
                ddlTemplate.Items.Clear();
                ddlTemplate.Items.Insert(0, new ListItem("-- Error Loading Templates --", "0"));
                Response.Write("Error loading templates: " + ex.Message);
            }
            finally
            {
                if (sqlconn.State == ConnectionState.Open)
                {
                    sqlconn.Close();
                }
            }
        }

        protected void ddlSupply1_SelectedIndexChanged(object sender, EventArgs e)
        {
            string supplierId = ddlSupply1.SelectedValue;

            if (supplierId == "false" || string.IsNullOrEmpty(supplierId))
            {
                lblCurrency.Text = "";
                return;
            }

            try
            {
                checkconnection();
                sqlconn.Open();

                string query = "SELECT Currency FROM Supplier WHERE SupplierId = @SupplierId";
                SqlCommand cmd = new SqlCommand(query, sqlconn);
                cmd.Parameters.AddWithValue("@SupplierId", supplierId);

                object result = cmd.ExecuteScalar();

                if (result != null)
                {
                    currency = result.ToString();
                    lblCurrency.Text = currency;
                    LoadNextSuppliers();
                }
            }
            catch (Exception ex)
            {
                lblCurrency.Text = "Error: " + ex.Message;
            }
            finally
            {
                sqlconn.Close();
            }
            LoadTemplates(total);
        }

            INSERT INTO Document (

                @Quotation, @Configuration, @CostBeakdown, @TemplateId,@ConfirmedBy

            SELECT SCOPE_IDENTITY();"; // Get the ID of the newly inserted row

                using (SqlCommand cmd = new SqlCommand(insertDocumentQuery, sqlconn))

                    cmd.Parameters.AddWithValue("@CompanyId", invoiceCompany);

        // Helper method to encapsulate saving logic for a single row
        private void SaveRecordData(int recordNumber, int DocumentID, string supplierId, string description, string quantity, string unitPrice, string total)
        {
            // Implement  database saving logic here for a single record row
            // Use the parameters passed in.
            try
            {
                checkconnection();
                sqlconn.Open();
                string query = @"INSERT INTO RequestedItemPayments 
(UnitPrice, Qty, Description, DocumentID, SupplierId) VALUES 
(@UnitPrice,@Quantity,@Description,@documentid,@SupplierId)";
                SqlCommand cmd = new SqlCommand(query, sqlconn);
                cmd.Parameters.AddWithValue("@documentid", DocumentID);
                cmd.Parameters.AddWithValue("@SupplierId", supplierId);
                cmd.Parameters.AddWithValue("@Description", description);
                cmd.Parameters.AddWithValue("@Quantity", quantity);
                cmd.Parameters.AddWithValue("@UnitPrice", unitPrice);
                cmd.ExecuteNonQuery();
            }
            catch (Exception ex)
            {
                // Handle error appropriately (log it, show message)
                Response.Write($"Error saving record {recordNumber}: " + ex.Message);
            }
            finally
            {
                if (sqlconn.State == ConnectionState.Open)
                {
                    sqlconn.Close();
                }
            }

        }
